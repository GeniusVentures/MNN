    HEADER / GATE:
    #ifdef MNN_SUPPORT_TRANSFORMER_FUSE
    Includes: all standard SGFP4 test includes plus "sgfp4_encode.hpp", "SGFP4RealShapeFixtures.h"
    using namespace MNN::Express;

    CONSTANT: constexpr float kVkEncodeRelTol = 1e-4f;

    METHOD runVulkanFixtureParity(bool paddedOnly) → bool:
    For each kRealShapeFixtures[i]:
      bool isPadded = (((f.dimO+63)/64)*64 != f.dimO || ((f.dimI+63)/64)*64 != f.dimI);
      if (paddedOnly &amp;&amp; !isPadded) continue;
      if (!paddedOnly &amp;&amp; isPadded) continue;

      1. container = sgfp4_encode::encode(f.inputWeights, f.dimO, f.dimI); if empty fail
      2. sidecarPath = sgfp4_test::tempPath("sgfp4_vk_encode_parity_" + std::to_string(i) + ".bin")
         Write container bytes to sidecarPath (reuse the binary-write helper from SGFP4TestUtil.hpp or write directly via std::ofstream binary)
      3. Build SGFP4Dequant op T (mirror SGFP4VulkanDequantTest.cpp pattern exactly):
           op->type = MNN::OpType_SGFP4Dequant
           op->main.type = MNN::OpParameter_SGFP4DequantParam
           param->magic = MNN::kSGFP4Magic
           param->external = {0, static_cast&lt;int64_t&gt;(container.size())}
           param->dims = {f.dimO, f.dimI}
           op->externalPath = sidecarPath
      4. output = Variable::create(Expr::create(op.get(), {}))
         buffer = Variable::save({output})
         net = Interpreter::createFromBuffer(buffer.data(), buffer.size())
         config.type = MNN_FORWARD_VULKAN
         session = net->createSession(config)
         net->runSession(session)
         outTensor = net->getSessionOutput(session, nullptr)
         outPtr = outTensor->host&lt;float&gt;()
         outCount = (int)outTensor->elementSize()
      5. MNNTEST_ASSERT(outCount == f.dimO * f.dimI) — must be true dims, not padded
      6. MNNTEST_ASSERT(checkVectorByRelativeError&lt;float&gt;(outPtr, f.expected, outCount, kVkEncodeRelTol))
      7. Clean up sidecarPath

    CLASS SGFP4VulkanEncodeParityTest:
    virtual bool run(int precision) {
      auto vulkanCreator = MNN::MNNGetExtraRuntimeCreator(MNN_FORWARD_VULKAN);
      if (nullptr == vulkanCreator) {
        MNN_PRINT("Vulkan backend not available — skipping SGFP4 Vulkan encode parity test\n");
        return true;
      }
      if (!runVulkanFixtureParity(false)) return false;  // aligned shapes only in Task 1
      return true;
    }
    MNNTestSuiteRegister(SGFP4VulkanEncodeParityTest, "op/sgfp4/vulkan_encode_parity");
    #endif // MNN_SUPPORT_TRANSFORMER_FUSE
  </action>
