<objective>
Implement the CPU decode-parity test suite for the C++ encoder (D-04/D-08 CPU leg). The suite encodes the kRealShapeFixtures input weights via sgfp4_encode::encode(), decodes the resulting container via dequant_sgfp4_container_cpu (or _crop for padded shapes), and asserts parity with the Python-encoded decoded reference at rtol 1e-4.

Purpose: primary always-on parity gate for the encoder; establishes decode-parity for both aligned and padded non-aligned shapes before the Vulkan leg (Plan 09-05) builds on the same fixture data.
Output: test/op/SGFP4EncodeTest.cpp registered as "op/sgfp4/encode".

    Create test/op/SGFP4EncodeTest.cpp with the following structure:

    HEADER / GATE:
    #ifdef MNN_SUPPORT_TRANSFORMER_FUSE (matches all other SGFP4 test files — required gate)
    Includes: &lt;cstdint&gt;, &lt;vector&gt;, "MNN/SGFP4DequantUtils.hpp", "MNNTestSuite.h", "TestUtils.h", "SGFP4TestUtil.hpp", "SGFP4RealShapeFixtures.h", "sgfp4_encode.hpp"
    
    CONSTANT:
    constexpr float kEncodeRelTol = 1e-4f;  // D-04 decode-vs-decode bar

    HELPER: paddedDim(int d) — inline: return ((d + 63) / 64) * 64;
    HELPER: isAligned(int dimO, int dimI) — return paddedDim(dimO)==dimO &amp;&amp; paddedDim(dimI)==dimI;

    METHOD testFixtureParity() → bool:
    Loop i in 0..kRealShapeFixtureCount:
      const auto&amp; f = kRealShapeFixtures[i];
      auto cppContainer = sgfp4_encode::encode(f.inputWeights, f.dimO, f.dimI);
      if (cppContainer.empty()) { MNN_ERROR("encode returned empty for %s\n", f.name); return false; }
      std::vector&lt;float&gt; cppOut(f.expectedCount, 0.f);
      bool ok;
      if (isAligned(f.dimO, f.dimI)) {
        ok = MNN::dequant_sgfp4_container_cpu(cppContainer.data(), cppContainer.size(), cppOut.data(), (int)f.expectedCount);
      } else {
        int pO = paddedDim(f.dimO), pI = paddedDim(f.dimI);
        ok = MNN::dequant_sgfp4_container_cpu_crop(cppContainer.data(), cppContainer.size(), cppOut.data(), f.dimO, f.dimI, pO, pI);
      }
      if (!ok) { MNN_ERROR("decode returned false for %s\n", f.name); return false; }
      if (!checkVectorByRelativeError&lt;float&gt;(cppOut.data(), f.expected, (int)f.expectedCount, kEncodeRelTol)) {
        MNN_ERROR("parity failed for %s\n", f.name); return false;
      }
    return true;

    METHOD testContainerFraming() → bool:
    Loop i in 0..kRealShapeFixtureCount:
      auto cppContainer = sgfp4_encode::encode(f.inputWeights, f.dimO, f.dimI);
      if (cppContainer.size() &lt; 16) return false;
      uint32_t magic; std::memcpy(&amp;magic, cppContainer.data(), 4);
      if (magic != MNN::kSGFP4Magic) { MNN_ERROR("bad magic\n"); return false; }
      if (cppContainer[4] != 0x02u) { MNN_ERROR("bad version\n"); return false; }
    return true;

    CLASS:
    class SGFP4EncodeTest : public MNNTestCase {
    public:
      virtual bool run(int precision) {
        if (!testContainerFraming()) return false;
        if (!testFixtureParity()) return false;
        return true;
      }
    };
    MNNTestSuiteRegister(SGFP4EncodeTest, "op/sgfp4/encode");

    #endif // MNN_SUPPORT_TRANSFORMER_FUSE
  </action>
